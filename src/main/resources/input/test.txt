#+ TEST INTERACTIVO - Cubre todas las operaciones soportadas por el compilador +#
#+ El programa le pide datos al usuario por teclado (Int y Float) y reacciona +#
#+ segun esos valores. Al final hay bloques comentados que documentan casos +#
#+ que deberian fallar (quedan comentados para no romper esta compilacion). +#

init {
    edad, limite, contador, opcion, iterador, valorCalculado, a, b, c, d : Int
    precio, cantidad, descuento : Float
    nombre : String
}

#+ --- STRING: asignacion y print --- +#
nombre = "Validador interactivo"
print(nombre)

#+ --- READ INTERACTIVO: Int --- +#
print("Ingrese su edad:")
read(edad)
print("Edad ingresada:")
print(edad)

print("Ingrese el limite para los bucles:")
read(limite)

#+ --- READ INTERACTIVO: Float --- +#
print("Ingrese el precio unitario:")
read(precio)

print("Ingrese la cantidad (admite decimales):")
read(cantidad)

print("Ingrese el descuento:")
read(descuento)

#+ --- ARITMETICA: +, -, *, / (mismo tipo a ambos lados) --- +#
total = precio * cantidad
total = total - descuento
print(total)

resultado = edad + limite
resultado = resultado - contador
resultado = resultado * 2
resultado = resultado / 2
print(resultado)

#+ --- CONSTANTES NEGATIVAS --- +#
contadorNegativo = -10
descuentoNegativo = -0.5
print(contadorNegativo)
print(descuentoNegativo)

#+ --- COMPARACIONES: >, >=, <, <=, ==, != --- +#
if (edad >= 18)
{
    print("Es mayor de edad")
}
else
{
    print("Es menor de edad")
}

if (precio > 100.0)
{
    print("Precio alto")
}
else if (precio == 100.0)
{
    print("Precio exacto")
}
else
{
    print("Precio bajo")
}

if (edad != limite)
{
    print("Edad y limite son distintos")
}

if (edad <= limite)
{
    print("Edad menor o igual al limite")
}

#+ --- IF ANIDADO --- +#
if (edad >= 18)
{
    if (precio > 0.0)
    {
        print("Mayor de edad con precio valido")
    }
}

#+ --- READ INTERACTIVO PARA NOT / AND / OR --- +#
print("Ingrese valor para A:")
read(a)
print("Ingrese valor para B:")
read(b)
print("Ingrese valor para C:")
read(c)
print("Ingrese valor para D:")
read(d)

if (a > b AND c > d)
{
    print("A mayor a B Y C mayor a D")
}

if (a > b OR c > d)
{
    print("A mayor a B O C mayor a D")
}

if (NOT (a > b))
{
    print("A no es mayor a B")
}

#+ --- WHILE CLASICO (usa el limite ingresado por teclado) --- +#
contador = limite
print("Comienzo de Ciclo WHILE Clasico:")
while (contador > 0)
{
    print(contador)
    contador = contador - 1
}

#+ --- WHILE ESPECIAL IN [ ... ] --- +#
print("Comienzo de Ciclo WHILE especial simple:")
while (opcion in [1, 2, 3])
{
    print(opcion)
}

print("Comienzo de Ciclo WHILE especial con operaciones:")
while (valorCalculado in [1+1, 2*2, 6-2])
{
    print(valorCalculado)
}

#+ --- FOR ... TO ... STEP ... NEXT (usa el limite ingresado) --- +#
print("Inicio de ciclo FOR desde 1 a Limite ingresado:")
FOR iterador = 1 TO limite STEP 1
    print(iterador)
NEXT iterador

#+ --- WHILE COMPUESTO CON REASIGNACION (usa A, B, C, D ingresados) --- +#
while (a < b AND c > d)
{
    print("Mientras A menor a B Y C mayor a D")
    a = b
    c = 2
}

print("Fin del programa interactivo")
